# Three minutes for four synthesised instruments.
#
# python -m gestate.audioperform examples/audio/quartet.ges
# python -m gestate.audioperform examples/audio/quartet.ges --seconds 20 -o intro.wav
#
# Both routes go through the compiled engine: without `-o` the piece
# plays in real time, and `-o` renders the whole three minutes in under a
# minute. Only a machine with no `clang` falls back to `audioengine` —
# one Python object per node per sample, about a thousandth of real time,
# roughly a day and a half for this file.
#
# `polysaw.ges` is one bank playing eight bars. `nocturne.ges` is a piece
# of some length handed to somebody else's synthesiser. This is both at
# once: **four `voices` banks, their own piece, and the mix**, in one file
# and therefore checked as one program — `voices.lead` names a bank
# lexically, so "does this line suit this instrument" is a type error rather
# than a silence.
#
# The four are chosen to be four *methods* rather than four timbres, so the
# file is also a tour of `synth.ges`:
#
# * `bass` — subtractive: a saw and a sub-octave square into a ladder
# whose cutoff an envelope sweeps.
# * `pad` — subtractive too, but slow, detuned and **stereo**: a voice
# returns a `Stereo`, so the bank sums left with left.
# * `lead` — four-operator FM, the wiring written as a matrix.
# * `kit` — hiss and a falling sine, with the *drum* in the payload:
# one voice function that is a kick, a snare or a hat depending on a
# number the score sends it.
#
# ── How a piece of this length is written ────────────────────────────────
#
# Every part below is a `[: Int :]` — bare numbers, no instrument — exactly
# as in `examples/music/`, and the bank is attached in one place at the
# bottom by `onPad`, `onBass` and friends. That is what lets a phrase be
# transposed, doubled at the octave, or moved to another instrument as a
# one-word edit, and it is why `shift` below is four words: `>>=`
# substitutes a score for every note, so "everything a fifth up" is a
# function of one number.
#
# 72 bars of 4/4 at 96 bpm is three minutes to the second. Nine sections
# of eight bars, and the one rule to keep in mind when editing is that
# everything inside a `||` starts at the same instant — so a section's four
# voices must agree on how many bars they are.
# ── What a note carries ──────────────────────────────────────────────────
#
# **One payload for all four banks**: a number and how hard. For the three
# pitched instruments the number is a key; for the kit it is which drum —
# 0 kick, 1 snare, 2 closed hat, 3 open hat.
#
# The kit would rather have a payload of its own, and saying `Hit` where
# the others say `Stroke` is exactly the kind of thing the type checker is
# here for. It does not build: two *instantiations of `Played`* in one
# graph collide in the code generator, which picks a layout by constructor
# tag and finds `Played` twice. Fifteen lines reproduce it — two banks,
# two records — so it is a defect rather than a rule of the fragment, and
# until it is fixed the banks share a record.
#
# It is not called `Note`: `music.ges` declares `Score a := Note a`, so a
# piece already has a `Note` constructor and `'x` desugars to it.
# `fmpoly.ges` says what shadowing it costs.
Stroke := Stroke Int Int # key number (or drum), velocity
voices bass 3 bassVoice : Sig Float
voices pad 8 padVoice : Sig Stereo
voices lead 8 leadVoice : Sig Float
voices kit 4 kitVoice : Sig Float
#: Every bank takes a `Stroke`, so a keyboard can play any of them against
#: the score — `--midi-bank pad`, which is `duet.ges`'s trick with three
#: more instruments to choose from.
instance FromMIDI Stroke where
noteOn ch p v = Just (Stroke p v)
# ── What every voice reads off its note ──────────────────────────────────
#
# A `voices` bank hands each voice the note's *timing* (`Sig Gate`) and the
# note's *payload* (`Sig Stroke`) as two separate signals, and almost
# nothing needs both. These are the payload's readings; `!` lifts one over
# the signal wherever it is wanted.
hzOf : Stroke -> Float
hzOf q = case q of
Stroke k v -> keyHz k
velOf : Stroke -> Float
velOf q = case q of
Stroke k v -> toFloat v / 127.0
#: A frequency signal moved by `cents`. A detune is a *factor*, and this
#: is `centsHz` lifted over a signal so that it can be one.
detuned : Float -> Sig Float -> Sig Float
detuned cents hz = map (h => centsHz cents h) hz
#: An envelope scaled into a cutoff in hertz, clamped at both ends.
#:
#: Every subtractive voice below wants this and none of them wants it
#: differently: a filter asked for a cutoff past Nyquist should sound like
#: an open filter rather than like a divergence.
sweepOf : Float -> Float -> Sig Float -> Sig Float
sweepOf base range env =
map (x => clamp 20.0 nyquist (base + range * x)) env
#: Where a key sits in the field: middle C centred, three octaves either
#: side reaching the edges.
keyPan : Stroke -> Float
keyPan q = case q of
Stroke k v -> clamp (negate 1.0) 1.0 (toFloat (k - 60) / 36.0)
# ── The bass: subtractive, with a sub-octave ─────────────────────────────
#
# A sawtooth at the note, a square an octave *below* it, and a four-pole
# ladder that opens fast and shuts again. The square is what makes it
# carry on a small speaker: the fundamental of a bass note at 73 Hz is
# mostly inaudible there, and its octave is not.
#
# **Three lines, and each is a value.** The sub-octave is `hz * 0.5` —
# arithmetic on the frequency signal, not a second state record — and the
# sweep is the filter envelope scaled into hertz.
bassQ : Float
bassQ = 0.42
bassAmpEnv : Adsr
bassAmpEnv = Adsr 0.004 0.22 0.72 0.12
#: Shorter than the amplitude envelope, so the note goes dull while it is
#: still loud — the shape of a plucked string, and the reason two envelopes
#: are worth having.
bassFiltEnv : Adsr
bassFiltEnv = Adsr 0.002 0.16 0.22 0.10
bassOsc : Sig Float -> Sig Float
bassOsc hz = 0.6 * saw hz + 0.3 * square (hz * 0.5)
bassVoice : Sig Gate -> Sig Stroke -> Sig Float
bassVoice g s = lowpassLadder (sweepOf 120.0 2200.0 (adsr bassFiltEnv g)) (!bassQ)
(bassOsc (!hzOf s))
* adsr bassAmpEnv g * !velOf s * 0.9
# ── The pad: slow, detuned, and two channels wide ────────────────────────
#
# `stereopad.ges`'s voice, and its reasoning: a voice is panned by its own
# *key number*, so a chord lays itself across the field and the position of
# a note is a property of the note rather than of which of the eight slots
# the allocator happened to hand it. The two detuned oscillators are then
# placed a little apart from each other inside the voice, which is the
# second half of the width.
#
# Eight voices for four-note chords is not extravagance: the release is
# over a second long, so a chord is still sounding while the next one
# starts, and a bank of four would steal its own tail every bar.
padDetune : Float
padDetune = 8.0
padSpread : Float
padSpread = 0.5
#: It arrives rather than starts.
padAmpEnv : Adsr
padAmpEnv = Adsr 0.45 1.4 0.55 1.1
padFiltEnv : Adsr
padFiltEnv = Adsr 1.1 1.8 0.4 0.9
#: Triangle and saw rather than two of either: the triangle carries the
#: body and the saw the top, and the filter takes back most of the saw.
padOsc : Sig Float -> Sig Float
padOsc hz = 0.55 * triangle (detuned (negate padDetune) hz)
+ 0.3 * saw (detuned padDetune hz)
#: The filtered, enveloped voice placed twice — once at each end of
#: `padSpread`, with the note's own pitch sliding the pair as a whole.
#:
#: Both oscillators are in the filter's output as one number, so they
#: cannot be panned apart afterwards; what is placed twice is the
#: *filtered* signal, which is what makes the width survive the filter.
place : Sig Float -> Sig Float -> Sig Stereo
place x p = zip (v c => panOf (c - padSpread * 0.5) v
+ panOf (c + padSpread * 0.5) v) x p
padVoice : Sig Gate -> Sig Stroke -> Sig Stereo
padVoice g s = place (lowpassSvf (sweepOf 220.0 2600.0 (adsr padFiltEnv g)) 0.3
(padOsc (!hzOf s))
* adsr padAmpEnv g * !velOf s * 0.3)
(!keyPan s)
# ── The lead: FM, wired as a matrix ──────────────────────────────────────
#
# Two pairs. Operators 1 and 2 are the body at a 1:1 ratio, so the tone is
# harmonic and reads as a pitched note; 3 and 4 are a struck-metal
# transient at 9×, which is not a whole-number multiple of anything else
# here. The diagonal entry on row 2 is operator 2 modulating itself —
# feedback needs no separate mechanism, because every operator already
# reads the previous sample's outputs.
#
# Turn `0.16` to `0.0` and the attack goes soft; turn `9.0` to `2.0` and
# the instrument becomes a horn.
bellPatch : Patch
bellPatch = Patch (Quad 1.0 1.0 1.0 9.0)
(modulates 2 1 1.05 (modulates 2 2 0.16
(modulates 4 3 0.5 noWiring)))
(Quad 0.8 0.0 0.3 0.0)
#: Four envelopes, one per operator, and that is what makes it an
#: instrument rather than a chord of sine waves: the modulators are faster,
#: so the *brightness* decays ahead of the loudness the way every struck
#: string does.
bodyEnv : Adsr
bodyEnv = Adsr 0.006 1.1 0.45 0.35
bodyModEnv : Adsr
bodyModEnv = Adsr 0.002 0.55 0.10 0.2
strikeEnv : Adsr
strikeEnv = Adsr 0.0 0.3 0.0 0.1
strikeModEnv : Adsr
strikeModEnv = Adsr 0.0 0.13 0.0 0.06
#: The four operator levels as one signal. `adsr` gives one envelope
#: from a gate; this is the same fold giving four at once, which is what a
#: `Quad` is and what saves three passes over the same two integers.
leadLevels : Sig Gate -> Sig Quad
leadLevels g = zip (t w => case w of
Gate on off -> Quad (adsrOf bodyEnv t on off) (adsrOf bodyModEnv t on off)
(adsrOf strikeEnv t on off)
(adsrOf strikeModEnv t on off)) ticks g
#: The bank is fed the note's frequency and those four levels — and that
#: is the whole voice. There used to be a `leadDrive` between here and
#: `fm` whose only job was to `zip` the two together; `fm` does its own
#: plumbing now, as the filters always have.
leadVoice : Sig Gate -> Sig Stroke -> Sig Float
leadVoice g s = fm bellPatch (!hzOf s) (leadLevels g) * !velOf s * 0.6
# ── The kit: one voice that is four drums ────────────────────────────────
#
# The payload's first field is *which* drum, and the voice sums four
# drums each multiplied by whether this note is that drum. A gate is a
# number here rather than a branch, which is what lets the whole thing stay
# an expression — and the three that are not playing are multiplied by
# zero, so a kick costs a kick.
#
# Nothing here is a sample. A kick is a sine whose pitch falls, a snare is
# hiss through a bandpass, and a hat is the same hiss through a highpass.
#
# **The hiss is one signal, shared.** Four voices each folding their own
# generator from the same seed produced four identical sequences, which is
# what `noise` is: a fold over time, so that a render is reproducible.
hiss : Sig Float
hiss = white 20260806
#: A kick is a pitch envelope before it is an amplitude envelope: 178 Hz
#: down to 48 in about forty milliseconds is the whole sound.
kick : Sig Gate -> Sig Float
kick g = sine (48.0 + 130.0 * perc 26.0 g) * perc 7.0 g * 0.95
snare : Sig Gate -> Sig Float
snare g = bandpassSvf 1900.0 0.45 hiss * perc 15.0 g * 0.8
#: The two hats are the same filter and differ only in how long they ring,
#: which is what a pedal does to a real one.
closedHat : Sig Gate -> Sig Float
closedHat g = highpassSvf 7400.0 0.18 hiss * perc 55.0 g * 0.3
openHat : Sig Gate -> Sig Float
openHat g = highpassSvf 7400.0 0.18 hiss * perc 10.0 g * 0.26
#: One per drum: this note's velocity if it is that drum, and zero if it is
#: not. `perc` is already silent before a note and needs no `off`, so
#: an untouched voice is silent on all four counts.
isKick : Stroke -> Float
isKick q = case q of
Stroke d vel -> case d of
0 -> toFloat vel / 127.0
_ -> 0.0
isSnare : Stroke -> Float
isSnare q = case q of
Stroke d vel -> case d of
1 -> toFloat vel / 127.0
_ -> 0.0
isClosed : Stroke -> Float
isClosed q = case q of
Stroke d vel -> case d of
2 -> toFloat vel / 127.0
_ -> 0.0
isOpen : Stroke -> Float
isOpen q = case q of
Stroke d vel -> case d of
3 -> toFloat vel / 127.0
_ -> 0.0
kitVoice : Sig Gate -> Sig Stroke -> Sig Float
kitVoice g s = kick g * !isKick s
+ snare g * !isSnare s
+ closedHat g * !isClosed s
+ openHat g * !isOpen s
# ── Shape ────────────────────────────────────────────────────────────────
#
# Repetition is a function rather than a copy, so the chord progression
# below is written once and played through fifteen times.
twice : [: a :] -> [: a :]
twice x = x ++ x
four : [: a :] -> [: a :]
four x = twice (twice x)
#: A bar of silence. `r` is one ticksPerBeat, like a note, and `|* 4` is what
#: makes it a bar — which means `barRest |* 4` is four bars.
barRest : [: a :]
barRest = r |* 4
rest4 : [: a :]
rest4 = barRest |* 4
#: Everything `n` semitones up. Four words, because `>>=` substitutes a
#: score for every unassigned note and a transposed note is a score.
shift : Int -> [: Int :] -> [: Int :]
shift n s = s >>= (k => '(k + n))
down : [: Int :] -> [: Int :]
down s = shift (0 - 12) s
# ── Harmony ──────────────────────────────────────────────────────────────
#
# D minor. Dm - Bb - F - C is the piece; Gm - Bb - C - A is the bridge,
# and the A is major, which is the one chord with an F# in it and the only
# reason the return to Dm afterwards sounds like a return.
#
# The voicings move by step rather than by root, so the inner notes barely
# shift between chords — which is what keeps eight slow voices from
# sounding like eight separate events.
padDm : [: Int :]
padDm = ('50 || '57 || '62 || '65) |* 4
padBb : [: Int :]
padBb = ('46 || '58 || '62 || '65) |* 4
padF : [: Int :]
padF = ('53 || '57 || '60 || '65) |* 4
padC : [: Int :]
padC = ('48 || '55 || '60 || '64) |* 4
padGm : [: Int :]
padGm = ('43 || '55 || '58 || '62) |* 4
padA : [: Int :]
padA = ('45 || '57 || '61 || '64) |* 4
padMain : [: Int :]
padMain = padDm ++ padBb ++ padF ++ padC
padBridge : [: Int :]
padBridge = padGm ++ padBb ++ padC ++ padA
# ── Bass ─────────────────────────────────────────────────────────────────
#
# One figure, given a root: two eighths up the octave, a quarter, two
# eighths to the fifth, and the octave to finish. Four beats however low
# the root is, so the whole line is `bassFig` applied four times.
bassFig : Int -> [: Int :]
bassFig k = (('k ++ '(k + 12)) |/ 2)
++ 'k
++ (('k ++ '(k + 7)) |/ 2)
++ '(k + 12)
bassMain : [: Int :]
bassMain = bassFig 38 ++ bassFig 34 ++ bassFig 41 ++ bassFig 36
bassBridge : [: Int :]
bassBridge = bassFig 31 ++ bassFig 34 ++ bassFig 36 ++ bassFig 33
#: The same roots as whole notes, for the sections where the drums are out
#: and the figure would be too busy.
bassLong : [: Int :]
bassLong = ('38 |* 4) ++ ('34 |* 4) ++ ('41 |* 4) ++ ('36 |* 4)
# ── Drums ────────────────────────────────────────────────────────────────
#
# 0 is the kick, 1 the snare, 2 the closed hat and 3 the open one. Hats on
# the eighths, so `|/ 2` halves a note and eight of them fill the bar the
# four backbeats do.
hatBar : [: Int :]
hatBar = twice (four ('2 |/ 2))
#: The last eighth of the bar opened, which is what makes a straight
#: pattern breathe at the bar line.
hatBarOpen : [: Int :]
hatBarOpen = twice (four ('2 |/ 2)) || at (7 * ticksPerBeat / 2) ('3 |/ 2)
backbeat : [: Int :]
backbeat = '0 ++ '1 ++ '0 ++ '1
#: The same, with the kick doubled into the third ticksPerBeat.
backbeatPush : [: Int :]
backbeatPush = '0 ++ '1 ++ (('0 ++ '0) |/ 2) ++ '1
groove : [: Int :]
groove = backbeat || hatBar
groovePush : [: Int :]
groovePush = backbeatPush || hatBar
#: Four snares in the space of one ticksPerBeat.
fill : [: Int :]
fill = four '1 |/ 4
#: `||` rather than `++`, so the bar stays a bar: sequencing the fill on
#: would make the phrase five beats long and every section after it would
#: start late. `at` places it inside the bar instead.
grooveFill : [: Int :]
grooveFill = groovePush || at (3 * ticksPerBeat) fill
drumPhrase : [: Int :]
drumPhrase = groove ++ groovePush ++ groove ++ grooveFill
#: Eight bars of hats alone, with the fill arriving at the end — what the
#: break needs so the return has something to arrive against.
hatsPhrase : [: Int :]
hatsPhrase = four hatBar
++ hatBar ++ hatBarOpen ++ hatBar
++ (hatBarOpen || at (3 * ticksPerBeat) fill)
# ── The tune ─────────────────────────────────────────────────────────────
#
# Quarters in the first phrase, eighth-note runs in the second and a climb
# in the third, so the return of the first in the reprise sounds like a
# return rather than like more of the same.
leadA1 : [: Int :]
leadA1 = '74 ++ '72 ++ ('74 |* 2)

++ '72 ++ '74 ++ ('69 |* 2)

leadA2 : [: Int :]
leadA2 = (('74 ++ '76) |/ 2) ++ '77 ++ ('76 |* 2)
++ (('77 ++ '79) |/ 2) ++ '81 ++ ('79 |* 2)
++ '77 ++ '76 ++ '74 ++ '72
++ ('76 |* 2) ++ '74 ++ r
leadA : [: Int :]
leadA = leadA1 ++ leadA2
#: The same eight bars with the runs turned into sixteenths — the verse
#: repeating without repeating itself.
leadA3 : [: Int :]
leadA3 = (('74 ++ '72 ++ '74 ++ '76) |/ 4) ++ ('74 |* 3)

leadBridge : [: Int :]
leadBridge = '70 ++ '69 ++ ('67 |* 2)

leadC : [: Int :]
leadC = (('74 ++ '77 ++ '81 ++ '79) |/ 2) ++ ('81 |* 2)

# ── Arpeggios ────────────────────────────────────────────────────────────
#
# Four chord tones, up and back, as eighths. The same bank as the tune —
# an arpeggio is not another instrument, it is what this one does when
# there is no melody to play.
arpBar : Int -> Int -> Int -> Int -> [: Int :]
arpBar a b c d = ('a ++ 'b ++ 'c ++ 'd ++ 'c ++ 'b ++ 'c ++ 'd) |/ 2
arpMain : [: Int :]

++ arpBar 55 60 64 67

# ── Instruments ──────────────────────────────────────────────────────────
#
# Where a part stops being numbers and becomes an instrument. `voices.pad`
# names the bank lexically, and the result is `[: Void :]`: every note
# committed, and the type checker will not let a `Stroke` reach the pad.
pitched : Int -> [: Int :] -> [: Stroke :]
pitched v s = s >>= (k => '(Stroke k v))
onPad : [: Int :] -> [: Void :]

onBass : [: Int :] -> [: Void :]

onLead s = pitched 100 s >>= voices.lead
onArp : [: Int :] -> [: Void :]

onKit : [: Int :] -> [: Void :]
onKit s = (s >>= (d => '(Stroke d 100))) >>= voices.kit

# ── The mix ──────────────────────────────────────────────────────────────
#
# Each bank's name is bound to the *sum* of its voices, so the mix is
# ordinary signal arithmetic. `pad` is already a `Sig Stereo` — its voices
# are — and the other three are placed with `pan` and `widen`, which
# is why `sound` can be stereo without any bank being asked to care.
#
# `drive` on the bass and on the master is a soft clipper: twenty-one
# voices summed pass 1.0 on a dense bar, and an analogue desk rounds those
# peaks off rather than tearing.
padMix : Sig Stereo
padMix = 0.6 * pad
bassMix : Sig Stereo
bassMix = widen (drive 0.05 (gain 0.42 bass))
leadMix : Sig Stereo
leadMix = pan 0.22 (dcBlock (lowpassSvf 7000.0 0.15 (gain 0.45 lead)))
kitMix : Sig Stereo
