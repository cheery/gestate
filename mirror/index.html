# mirror.ges — a piece that watches itself.
#
# python -m gestate.audioperform examples/audio/mirror.ges --seconds 40 -o take.wav
# python -m gestate.online examples/audio/mirror.ges -o site/
#
# **Written for the gallery**, which is what makes it different from its
# neighbours: `lantern.ges` shows a peak, `spectrum.ges` shows eight
# bands and `scoped.ges` shows a trace, and no file had ever shown all
# three at once. This one does, and every one of them is the *host*
# reaching the picture — the direction that costs a program nothing to
# ask for and that a tab could not do until 2026-09-04
# (`card:audiovisual-gallery.md`).
#
# **There are no faders on it, on purpose.** A fader on this page would
# move the picture and not the sound: the canvas-to-parameter bridge is
# the plugin's and the tab has none yet. So the face here is a
# *readout* and nothing on it pretends to be a control — every pixel is
# something the instrument said. When the bridge lands, faders belong
# on `lantern.ges`, which already has them.
#
# The bare `canvas` line asks for the picture as a live content box on
# that line in the workbench; delete it and the box goes.
canvas
# ── What the instrument says back ────────────────────────────────────────
#
# Three readings, three shapes, and a file is charged only for the ones
# it declares: the trace is the window the scope's own ring already
# holds, the bands come from a filter bank the host runs per sample, and
# the peak is the loudest sample since the picture last looked.
#: The waveform. 128 points, the scope's window downsampled by
#: max-absolute per bucket — a scope that averaged away a click would be
#: a scope that lies (`spec/scope.md`).
trace : Chan (List Float)
trace = chan
wave : Sig (List Float)
wave = Nil ::: mkSig (wait trace)
#: The spectrum, low to high. Declaring `band0` is what switches the
#: bank on; a file that does not is not charged for it.

band0 = chan
band1 : Chan Float
band1 = chan
band2 : Chan Float
band2 = chan
band3 : Chan Float
band3 = chan
band4 : Chan Float
band4 = chan
band5 : Chan Float
band5 = chan
band6 : Chan Float
band6 = chan
band7 : Chan Float
band7 = chan
level0 : Sig Float
level0 = 0.0 ::: mkSig (wait band0)
level1 : Sig Float
level1 = 0.0 ::: mkSig (wait band1)
level2 : Sig Float
level2 = 0.0 ::: mkSig (wait band2)
level3 : Sig Float
level3 = 0.0 ::: mkSig (wait band3)
level4 : Sig Float
level4 = 0.0 ::: mkSig (wait band4)
level5 : Sig Float
level5 = 0.0 ::: mkSig (wait band5)
level6 : Sig Float
level6 = 0.0 ::: mkSig (wait band6)
level7 : Sig Float
level7 = 0.0 ::: mkSig (wait band7)
#: How loud, since the last frame.

loud = 0.0 ::: mkSig (wait peak)
# ── What you hear ────────────────────────────────────────────────────────
#
# Something with a **moving spectrum and a legible waveform**, because
# those are the two things the face is showing and a piece that sat still
# would be a demonstration of nothing. A plucked lead climbing through a
# filter that opens as it goes, a bass that stays put, and a little air
# so the top bars have something to say.
#: A note is a degree and a weight — the score picks the step, the voice
#: decides what it sounds like.
Tone := Tone Float Int
#: A minor pentatonic, which keeps a drawn phrase consonant however the
#: dice fall. Degrees climb by octaves past the table's end.
stepOf : Int -> Int
stepOf d = case d of
0 -> 0
1 -> 3
2 -> 5
3 -> 7
4 -> 10
5 -> 12
6 -> 15
_ -> 17
hzOf : Tone -> Float
hzOf t = case t of
Tone v d -> keyHz (45 + stepOf d)
velOf : Tone -> Float
velOf t = case t of
Tone v d -> v
lowOf : Tone -> Float
lowOf t = case t of
Tone v d -> keyHz (21 + stepOf d)
#: **The filter opens with the note's own pitch**, so a phrase climbing
#: the scale sweeps the bank rather than lighting one bar and holding it.
#: That is the whole reason this voice is shaped the way it is: the
#: picture is of the sound, so the sound has to give the picture
#: something to draw.
brightOf : Tone -> Float
brightOf t = case t of
Tone v d -> 320.0 + 260.0 * toFloat (stepOf d)
leadVoice : Sig Gate -> Sig Tone -> Sig Float
leadVoice g s =
lowpassSvf (!brightOf s) 0.30 (saw (!hzOf s))
* (!velOf s) * 0.30 * perc 6.0 g
#: The ground: two octaves down, round, slow to fade — the bottom two
#: bars, and something for the trace to sit on.
bassVoice : Sig Gate -> Sig Tone -> Sig Float
bassVoice g s =
lowpassSvf 240.0 0.14 (triangle (!lowOf s) * 0.55)
* (!velOf s) * perc 1.6 g
voices lead 5 leadVoice : Sig Float
voices bass 2 bassVoice : Sig Float
#: A hiss, quiet, so the top two bands are not dead the whole time.
air : Sig Float
air = 0.035 * highpassOnePole 5200.0 (white 5)
#: **`scope` is where the trace comes from** — it keeps a window of the
#: samples flowing through it and the engine publishes the ring. It is a
#: pass-through: the sound is unchanged by being watched.
sound : Sig Float
sound = scope "trace" (brickwall 0.92 (reverb 0.55 0.22 (lead + bass) + air))
# ── What it plays ────────────────────────────────────────────────────────
#: Four figures, drawn a bar at a time. They climb, because a phrase
#: that climbs walks the filter up the bank and the bars follow it.
figureOf : Int -> [: Tone :]
figureOf n = case n of
0 -> '(Tone 0.85 0) ++ '(Tone 0.70 2) ++ '(Tone 0.80 4) ++ '(Tone 0.65 6)
1 -> '(Tone 0.80 1) ++ '(Tone 0.75 3) ++ ('(Tone 0.90 5) |* 2)
2 -> ('(Tone 0.70 2) |* 2) ++ '(Tone 0.85 5) ++ '(Tone 0.60 7)
_ -> '(Tone 0.90 4) ++ '(Tone 0.60 3) ++ '(Tone 0.75 6)
++ (chance 0.5 '(Tone 0.95 7))
bar : [: Tone :]
bar = do
s <- draw
long 4 (figureOf (below 4 s))
#: The footing, and it never moves: a chancy piece with nothing fixed is
#: only weather.
footing : [: Tone :]
footing = ('(Tone 0.9 0) |* 2) ++ ('(Tone 0.8 2) |* 2)
score : [: Void :]
score = (cycle bar >>= voices.lead) || (cycle footing >>= voices.bass)
bpm : Int
bpm = 96
# ── What you see ─────────────────────────────────────────────────────────
#
# A readout in three parts, stacked: the waveform across the top, the
# spectrum under it, and the level down the right of the bars. Nothing
# below mentions time — they are ordinary functions to a `Sub`, lifted
# over their signals with `!` at the very end.
night : Colour
night = RGB 26 29 36
dim : Sig Colour
dim = colour 118 126 140
# ── the waveform ────
traceW : Int
traceW = 384
traceH : Int
traceH = 96
#: One point of the trace, as a short bar hanging from the centre line at
#: its own value's height. Placed by its centre like everything else, so
#: the height is where the value says and the bar is what makes a thin
#: waveform readable at a glance.
#:
#: **Clamped**, because a sample may exceed the height the picture
#: reserved and a picture that grew with the music would move the
#: spectrum underneath it.
pointAt : Int -> Float -> Sub
pointAt i p = Shift (i * 3 - 192)
(floor (clamp (0.0 - 1.0) 1.0 (p * 2.4) * toFloat (traceH / 2)))
(Rect 2 3 (RGB 122 200 235))
#: Drawn by recursion over the list, the way `ladder.ges` counts keys.
#: An empty list is an empty picture, which is what a piece that is not
#: playing correctly shows.
pointsFrom : Int -> List Float -> Sub
pointsFrom i ps = case ps of
Nil -> Gap 0 0
p :: rest -> Over (pointAt i p) (pointsFrom (i + 1) rest)
#: The centre line, so the trace has something to be a deviation from —
#: and `Sized` pins the box whatever the wave is doing.
scopeOf : List Float -> Sub
scopeOf ps = Sized traceW traceH
(Over (Over (Rect traceW traceH night) (Rect traceW 1 (RGB 52 58 70)))
(pointsFrom 0 ps))
# ── the spectrum ────
barW : Int
barW = 30
barH : Int
barH = 120
#: A square root rather than the level: a band a tenth as loud is not a
#: tenth as interesting, and a linear bar spends most of its height on
#: the difference between loud and very loud.
heightOf : Float -> Int
heightOf v = floor (toFloat barH * clamp 0.0 1.0 (sqrt (v * 1.8)))
#: Cold at the bottom of the range, warm at the top, so which band is
#: loud reads by colour as well as by height.
hueOf : Int -> Colour
hueOf i = RGB (64 + 22 * i) (198 - 13 * i) (214 - 20 * i)
trough : Sub
trough = Rect barW barH (RGB 34 38 46)
barOf : Int -> Float -> Sub
barOf i v = Sized (barW + 6) barH (Over trough
(Shift 0 ((barH - heightOf v) / 2) (Rect barW (heightOf v) (hueOf i))))
bars : Sig Sub
bars = !barOf 0 level0 `row` !barOf 1 level1
`row` !barOf 2 level2 `row` !barOf 3 level3
`row` !barOf 4 level4 `row` !barOf 5 level5
`row` !barOf 6 level6 `row` !barOf 7 level7
# ── the level ────
#: Green while there is room, amber near the top, red at the last two —
#: and a segment above the level is drawn dark rather than left out, so
#: the eye reads a height instead of counting bars.
litOf : Int -> Colour
litOf i = case i > 5 of
True -> RGB 226 92 78
False -> case i > 3 of
True -> RGB 226 176 74
False -> RGB 92 198 132
segOf : Int -> Float -> Sub
segOf i v = case v * 8.0 > toFloat i of
True -> Rect 22 11 (litOf i)
False -> Rect 22 11 night
ladderOf : Int -> Float -> Sub
ladderOf i v = case i of
0 -> segOf 0 v
_ -> Column (segOf i v) (Column (Gap 22 4) (ladderOf (i - 1) v))
meterOf : Float -> Sub
meterOf v = Sized 26 barH (ladderOf 7 v)
#: A caption under the thing it names. The box is written down here and
#: the host fits the glyphs into it, so nothing in this file measures
#: text.
#: **The box decides how big the letters come out** — `label` fits them
#: into the room reserved, `min(w / (4n - 1), h / 5)` on a 3x5 cell — so
#: 100 wide draws `SPECTRUM`'s eight characters at 2 where 60 drew them
#: at 1. Asking for bigger letters *is* reserving more room.
caption : String -> Sig Sub
caption s = column (gap 0 7) (label 100 12 s dim)
#: **Each part captioned under itself**, not the row captioned as a
#: whole. `label` fits the glyphs into the box this file reserves —
#: `min(w / (4n - 1), h / 5)` on a 3x5 cell — so a long string in a
#: small box comes out at scale 1 and is unreadable. Sizing a caption
#: *is* setting its type size, and two short words in their own boxes
#: read where one long one did not.
readout : Sig Sub
readout = column bars (caption "SPECTRUM")
`row` gap 16 0
`row` column (!meterOf loud) (caption "PEAK")
substrate : Sig Sub
substrate =
column (column (!scopeOf wave) (caption "TRACE"))
(column (gap 0 12) readout)
