# Twinkle Twinkle Little Star — a first song, written in the room on
# 2026-08-28 for Michael, who had just heard the beginner piece and asked
# how he would write his own.
#
# python -m gestate.audioperform examples/audio/twinkle.ges -o twinkle.wav
# python -m gestate.workbench examples/audio/twinkle.ges
#
# The instrument (top) is lesson 1 of doc/intermediate.md unchanged.
# The song (bottom) is the part to play with: `n 60` is one beat of
# middle C, `++` is "then", `|* 2` holds a note for two beats, and the
# tune is the three lines in the order the song goes.
Key := Key Int Int # key number, velocity
voices lead 4 leadVoice : Sig Float
instance FromMIDI Key where
noteOn ch p v = Just (Key p v)
env : Adsr
env = Adsr 0.01 0.2 0.6 0.35
hzOf : Key -> Float
hzOf q = case q of
Key k v -> keyHz k
velOf : Key -> Float
velOf q = case q of
Key k v -> toFloat v / 127.0
leadVoice : Sig Gate -> Sig Key -> Sig Float
leadVoice g s = triangle (!hzOf s) * adsr env g * !velOf s
# ── The song ─────────────────────────────────────────────────────────────
bpm : Int
bpm = 100
#: one note, one beat. 60 is middle C; each step of 1 is a semitone.
n : Int -> [: Key :]
n k = '(Key k 90)
#: "twinkle twinkle little star" C C G G A A G—
line1 : [: Key :]
line1 = n 60 ++ n 60 ++ n 67 ++ n 67 ++ n 69 ++ n 69 ++ n 67 |* 2
#: "how I wonder what you are" F F E E D D C—
line2 : [: Key :]
line2 = n 65 ++ n 65 ++ n 64 ++ n 64 ++ n 62 ++ n 62 ++ n 60 |* 2
#: "up above the world so high" G G F F E E D— (twice)
line3 : [: Key :]
line3 = n 67 ++ n 67 ++ n 65 ++ n 65 ++ n 64 ++ n 64 ++ n 62 |* 2
tune : [: Key :]
tune = line1 ++ line2 ++ line3 ++ line3 ++ line1 ++ line2
score : [: Void :]
score = tune >>= voices.lead
sound : Sig Float
sound = gain 0.3 lead
