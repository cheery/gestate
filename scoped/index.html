# scoped.ges — a scope on a playing signal, drawn — `spec/scope.md`.
#
#   python -m gestate.workbench examples/audio/scoped.ges
#
# Open the canvas and the trace is the sound: `scope "post"` keeps the
# window of samples flowing through it, the engine publishes the ring,
# and the picture below is an ordinary substrate folding the points
# into rects.  A scope is not a widget — it is a substrate somebody
# writes, and this file is the somebody.

#: The trace arrives here: 128 points, the scope's window downsampled
#: by max-absolute per bucket (`Workbench.TRACE_POINTS`).
post : Chan (List Float)
post = chan

#: Something worth watching: a slow wobble on the level, so the trace
#: breathes where a bare tone would stand still.
wobble : Sig Float
wobble = 0.5 + 0.4 * sine 0.4

sound : Sig Float
sound = scope "post" (0.3 * wobble * sine 220.0)

night : Colour
night = RGB 32 35 43

lit : Colour
lit = RGB 122 200 235

#: One dot per point, placed at its value's own height — a waveform,
#: drawn by recursion over the list like `ladder.ges` counts keys.
dots : Int -> List Float -> Sub
dots i ps = case ps of
    Nil -> Gap 0 0
    p :: rest -> Over (Shift (i * 3 - 192) (floor (p * 300.0))
                             (Rect 2 3 lit))
                      (dots (i + 1) rest)

face : List Float -> Sub
face ps = Over (Rect 400 140 night) (dots 0 ps)

substrate : Sig Sub
substrate = !face (Nil ::: mkSig (wait post))
