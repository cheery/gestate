# SPIRAL — one figure, turning faster each time round.
#
# python -m gestate.workbench examples/audio/spiral.ges
# python -m gestate.audioperform examples/audio/spiral.ges --seconds 32 -o spiral.wav
#
# The whole piece is one four-note figure and a number that says how
# finely to divide it. The number walks 1, 2, 2, 4 and back to 1, so the
# figure tightens for three turns and opens again — an accelerando you
# can hear the shape of, rather than one that dissolves.
#
# `unfold` is what makes that possible: the score is endless and each
# turn is written from the one before it, so nothing is a list of bars.
# The drone underneath is not in the score at all — it is arithmetic on
# the renderer's own clock, so it keeps time whatever the figure does.
bpm : Int
bpm = 96
#: Beats per second, which is the only place the tempo is written twice.
#: Everything with a rate reads it, so changing `bpm` moves the drone
#: and the hits with the notes instead of leaving them behind.
#:
#: A plain `Float`, lifted with `!` where a signal is wanted — so the
#: arithmetic on it (`pulseHz / 2`) happens once at the value level
#: rather than every sample.
pulseHz : Float
pulseHz = 1.6
# ── The mode ─────────────────────────────────────────────────────────────
#
# A natural minor on A, written out rather than computed: two octaves is
# fourteen numbers and a table you can read beats an expression you have
# to evaluate. **Every part reads pitch through here**, so the figure,
# the bass and the bell cannot drift apart — the same discipline
# `nightdrive.ges` uses for its chords.
step : Int -> Int
step d = case d of
0 -> 0
1 -> 2
2 -> 3
3 -> 5
4 -> 7
5 -> 8
6 -> 10
7 -> 12
8 -> 14
9 -> 15
10 -> 17
11 -> 19
12 -> 20
_ -> 22
#: A degree of the mode, as a key. A3 is the floor of the piece.
pitch : Int -> Int
pitch d = 57 + step d
# ── The turning ──────────────────────────────────────────────────────────
#
# Four turns to a lap. A turn names where in the mode the figure sits
# and how finely it divides — and both are read from the same index, so
# the spiral tightens *as* it climbs rather than doing the two things at
# unrelated times.
#: Which degree the figure starts on. Down a third, up a fourth, down a
#: third: it wanders and comes home, which is what keeps four turns from
#: sounding like a scale.
rootOf : Int -> Int
rootOf t = case t % 4 of
0 -> 0
1 -> 4
2 -> 2
_ -> 5
#: How many pieces the figure is cut into on this turn.
speedOf : Int -> Int
speedOf t = case t % 4 of
0 -> 1
1 -> 2
2 -> 2
_ -> 4
#: Up the triad and back down a step. Four notes, the same four every
#: turn — what changes is where they sit and how fast they go.
triad : Int -> [: Int :]
triad r = '(pitch r) ++ '(pitch (r + 2)) ++ '(pitch (r + 4)) ++ '(pitch (r + 3))
figure : Int -> [: Int :]
figure t = triad (rootOf t)
#: The bass, one note under the whole turn, two octaves down.
under : Int -> [: Int :]
under t = '(pitch (rootOf t) - 24)
# ── The score ────────────────────────────────────────────────────────────
#
# `unfold` grows one turn from the last, so the piece is endless without
# being a list. The turn index rides in the score as a *value* lasting
# four beats; binding it spreads the figure across exactly that span,
# which is why `speedOf` divides rather than shortens.
#: The figure, cut as finely as this turn asks.
plays : Int -> [: Void :]
plays t = (figure t |/ speedOf t) >>= voices.lead @ Note
#: And the ground under it, one note for the whole turn.
grounds : Int -> [: Void :]
grounds t = under t >>= voices.bass @ Note
score : [: Void :]
score = do
t <- unfold 0 (t => ('t |* 4, t + 1))
plays t || grounds t
# ── The voices ───────────────────────────────────────────────────────────
Note := Note Int
freq : Note -> Float
freq (Note x) = keyHz x
#: So the piece can be played from a keyboard as well as performed —
#: which is the point of the editor's piano, and costs one instance.
instance FromMIDI Note where
noteOn c n v = Just (Note n)
voices lead 8 leadVoice : Sig Float
voices bass 2 bassVoice : Sig Float
#: A pluck: saw for edge, triangle for body, and a filter that opens
#: with the envelope so the *start* of a note is the loud bright part.
#: The figure is the subject of the piece, so what its notes need is a
#: clear front edge rather than a rich tail.
#: **Scaled here rather than in the mix.** A bank is a *sum* — eight
#: voices of it — so a voice that peaks at one peaks at eight when the
#: figure is dense, and the limiter ends up doing the mixing. One voice
#: is quiet on purpose.
pluck : Sig Float -> Sig Float -> Sig Float
pluck hz env = 0.45 * lowpassSvf (300 + 4200 * env) 0.55
(saw hz * 0.5 + triangle hz * 0.5)
* env
leadVoice : Sig Gate -> Sig Note -> Sig Float
leadVoice gate note = pluck (!freq note) (perc 9 gate)
#: Round, and slow to arrive, so it reads as ground rather than as a
#: second voice competing with the figure.
bassVoice : Sig Gate -> Sig Note -> Sig Float
bassVoice gate note = 0.5 * lowpassSvf 400 0.7 (saw (!freq note))
* adsr (Adsr 0.02 0.3 0.7 0.4) gate
# ── The drone ────────────────────────────────────────────────────────────
#
# None of this is in the score. It is arithmetic on the renderer's
# clock, so it keeps time however the figure divides — the argument
# `nightdrive.ges` makes for its drums, and the reason the piece has a
# floor to stand on while the top of it accelerates.
#: A gate every `1 / n` of a second — the metronome the hits read.
every : Sig Float -> Sig Gate
every n = gateOn (!(x => x < 0.2) (phase n))
#: A saw that will not sit still, filtered by something that will not
#: either. Two slow movements at different rates never line up, which
#: is what keeps a drone from turning into a note.
grind : Sig Float
grind = saw (55.0 + 8 * sine 0.07) * unipolar (triangle 0.11)
bed : Sig Float
bed = 0.16 * lowpassSvf (700 + 500 * unipolar (sine 0.13)) 0.8 grind
#: Chopped on the beat, and only *half* chopped — the untouched copy
#: underneath is what stops the gaps from sounding like dropouts.
chop : Sig Float
chop = bed * (0.45 + 0.55 * unipolar (pulse 0.35 (!pulseHz)))
#: Air: a noise burst every two beats, through the same room the rest
#: of the piece is in.
breath : Sig Float
breath = 0.35 * white 7 * perc 26 (every (!(pulseHz / 2)))
#: And a tick on every beat, dry, to hold the grid.
tick : Sig Float
tick = 0.18 * pulse 0.25 (!(pulseHz * 220)) * perc 90 (every (!pulseHz))
# ── The mix ──────────────────────────────────────────────────────────────
#
# One room, and everything that should sound *in* it goes in together —
# two separate reverbs on two sources is two rooms, which is the sketch's
# mistake and the easiest one to hear.
wet : Sig Float
wet = 0.35 * reverb 0.6 0.02 (lead * 0.5 + breath)
#: `limit` is a seatbelt, not a mixer: the sum below is written to sit
#: near 0.8 on its own, so the limiter only catches the turns where the
#: figure is at its densest and every voice lands together.
sound : Sig Float
sound = limit 0.95 (dcBlock (chop + tick + bass * 0.35 + lead * 0.4 + wet))
